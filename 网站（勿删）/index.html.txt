        // 更新已登录用户的UI
        function updateUIForLoggedInUser() {
            // 隐藏访客菜单，显示用户菜单
            elements.guestMenu.classList.add('hidden');
            elements.userMenu.classList.remove('hidden');
            
            // 隐藏访客卡片，显示用户卡片
            elements.guestCard.classList.add('hidden');
            elements.userCard.classList.remove('hidden');
            
            // 更新用户信息
            document.querySelectorAll('.user-menu-btn img, .user-card img').forEach(img => {
                img.src = currentUser.avatar;
            });
            
            document.querySelectorAll('.user-menu-btn span').forEach(span => {
                span.textContent = currentUser.username;
            });
            
            // 更新用户卡片信息
            const userCard = elements.userCard;
            userCard.querySelector('h3').textContent = currentUser.username;
            userCard.querySelector('p').textContent = `积分: ${currentUser.level * 50} | 等级: ${currentUser.level}`;
            userCard.querySelectorAll('div > p.font-bold')[0].textContent = currentUser.posts;
            userCard.querySelectorAll('div > p.font-bold')[1].textContent = currentUser.followers;
            userCard.querySelectorAll('div > p.font-bold')[2].textContent = currentUser.following;
        }

        // 更新访客用户的UI
        function updateUIForGuestUser() {
            // 显示访客菜单，隐藏用户菜单
            elements.guestMenu.classList.remove('hidden');
            elements.userMenu.classList.add('hidden');
            
            // 显示访客卡片，隐藏用户卡片
            elements.guestCard.classList.remove('hidden');
            elements.userCard.classList.add('hidden');
        }

        // 显示提示消息
        function showToast(message) {
            // 创建提示元素
            const toast = document.createElement('div');
            toast.className = 'fixed top-20 left-1/2 transform -translate-x-1/2 bg-dark text-white px-6 py-3 rounded-lg shadow-lg z-50 animate-fadeIn';
            toast.textContent = message;
            
            // 添加到页面
            document.body.appendChild(toast);
            
            // 3秒后移除
            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transform = 'translate(-50%, -10px)';
                toast.style.transition = 'opacity 0.3s, transform 0.3s';
                
                setTimeout(() => {
                    document.body.removeChild(toast);
                }, 300);
            }, 3000);
        }
